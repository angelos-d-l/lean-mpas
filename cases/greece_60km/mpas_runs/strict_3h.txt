# (nvfortran -O2 -Kieee -Mnofma: MPAS-nv-strict and lean's ./lean build --fc nvfortran).

ranks 1

stream output output_interval 1:00:00
stream diagnostics output_interval 1:00:00
expect history.2026-11-08_03.00.00.nc
expect diag.2026-11-08_03.00.00.nc
